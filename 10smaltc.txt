    int i;

    return a + b;

    int result = 0;

Test 06 - 遞迴函式 (Fibonacci)

int fib(int n) {
    if (n <= 0) return 0;
    if (n == 1) return 1;
    return fib(n - 1) + fib(n - 2);
}

int main() {
    int i;
    printf("Fibonacci: ");
    for (i = 0; i < 7; i = i + 1) {
        printf("%d ", fib(i));
    }
    printf("\n");
    return 0;
}

預期輸出：
Fibonacci: 0 1 1 2 3 5 8 
程式回傳: 0

Test 08 - 指標操作

void swap(int *a, int *b) {
    int temp;
    temp = *a;
    *a = *b;
    *b = temp;
}

int main() {
    int x = 5;
    int y = 10;
    int *ptr;
    
    printf("Before: x = %d, y = %d\n", x, y);
    swap(&x, &y);
    printf("After: x = %d, y = %d\n", x, y);
    
    ptr = &x;
    printf("*ptr = %d\n", *ptr);
    return 0;
}

預期輸出：
Before: x = 5, y = 10
After: x = 10, y = 5
*ptr = 10
程式回傳: 0
